# 1. Mata todos los procesos de Docker Desktop
Stop-Process -Name "Docker Desktop" -Force -ErrorAction SilentlyContinue
Stop-Process -Name "com.docker.backend" -Force -ErrorAction SilentlyContinue
Stop-Process -Name "vmmem" -Force -ErrorAction SilentlyContinue

# 2. Reinicia la VM interna
Restart-VM -Name "DockerDesktopVM" -Force

# 3. Espera 15 segundos y relanza la GUI (opcional)
Start-Sleep -Seconds 15
Start-Process "C:\Program Files\Docker\Docker\Docker Desktop.exe"

# 4. Setea el socket correcto si usarás la CLI
$env:DOCKER_HOST = "npipe:////./pipe/docker_engine"
docker ps

Opción 2: Reinicio selectivo desde el task scheduler (más seguro)
Si no puedes o no quieres abrir GUI, crea un .bat o .ps1 con:
+++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++

taskkill /f /im "com.docker.backend"
timeout /t 10
net stop com.docker.service
net start com.docker.service
start "" "C:\Program Files\Docker\Docker\Docker Desktop.exe"
